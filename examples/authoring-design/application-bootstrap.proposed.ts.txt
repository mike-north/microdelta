import { readFile } from 'node:fs/promises';
import type { ICatalog } from './catalog-analysis.js';

/** Rejects unsupported fixture content at the application boundary. */
function isCatalog(value: unknown): value is ICatalog {
  if (typeof value !== 'object' || value === null ||
      !('complete' in value) || typeof value.complete !== 'boolean' ||
      !('pullRequests' in value) || !Array.isArray(value.pullRequests)) {
    return false;
  }
  return value.pullRequests.every((pr: unknown) =>
    typeof pr === 'object' && pr !== null &&
    'id' in pr && typeof pr.id === 'string' &&
    'title' in pr && typeof pr.title === 'string' &&
    'labels' in pr && Array.isArray(pr.labels) &&
    pr.labels.every((label: unknown) => typeof label === 'string'));
}
const parsed: unknown = JSON.parse(await readFile('./catalog.json', 'utf8'));
if (!isCatalog(parsed)) {
  throw new Error('Invalid catalog');
}
const snapshot = parsed;
